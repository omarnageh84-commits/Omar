window.applyTheme = function(name){
  try{
    var th = window.themes[name] || window.themes['forest'];
    var r = document.documentElement;
    for(var k in th){ if(th.hasOwnProperty(k)) r.style.setProperty(k, th[k]); }
    try{ document.body.style.background = th['--bg']; document.body.style.color = th['--text']; }catch(e){}
    try{ localStorage.setItem('omar_theme', name); localStorage.setItem('omar_theme_trigger', Date.now().toString()); }catch(e){}
  }catch(e){}
};
window.applyFontSize = function(size){
  try{
    var v = parseInt(size) || 13;
    if(v < 10) v = 10; if(v > 22) v = 22;
    // طريقة مزدوجة لضمان العمل في SPCK WebView
    document.documentElement.style.setProperty('--font-size-base', v + 'px');
    document.documentElement.style.fontSize = v + 'px';
    try{ document.body.style.fontSize = v + 'px'; }catch(e){}
    try{
      var style = document.getElementById('omar-font-override');
      if(!style){
        style = document.createElement('style');
        style.id = 'omar-font-override';
        document.head.appendChild(style);
      }
      style.textContent = 'html{font-size:'+v+'px !important;} :root{--font-size-base:'+v+'px !important;}';
    }catch(e){}
    try{ localStorage.setItem('omar_font_size', v.toString()); localStorage.setItem('omar_font_trigger', Date.now().toString()); }catch(e){}
    var sl = document.getElementById('fontSlider'); if(sl) sl.value = v;
    var fv = document.getElementById('fontVal'); if(fv) fv.textContent = v + 'px';
  }catch(e){ console.log('applyFontSize error', e); }
};
(function(){
  try{ var th = localStorage.getItem('omar_theme') || 'forest'; window.applyTheme(th); }catch(e){}
  try{ var fs = localStorage.getItem('omar_font_size') || '13'; window.applyFontSize(fs); }catch(e){}
})();
try{
  window.addEventListener('storage', function(e){
    if(e.key === 'omar_theme' && e.newValue) window.applyTheme(e.newValue);
    if(e.key === 'omar_theme_trigger'){ try{ var th = localStorage.getItem('omar_theme') || 'forest'; window.applyTheme(th); }catch(ex){} }
    if(e.key === 'omar_font_size' && e.newValue) window.applyFontSize(e.newValue);
    if(e.key === 'omar_font_trigger'){ try{ var fs = localStorage.getItem('omar_font_size') || '13'; window.applyFontSize(fs); }catch(ex){} }
  });
}catch(e){}
try{
  window.addEventListener('message', function(e){
    if(!e.data) return;
    if(e.data.type === 'omar_theme_change' && e.data.theme) window.applyTheme(e.data.theme);
    if(e.data.type === 'omar_font_change' && e.data.size) window.applyFontSize(e.data.size);
  });
}catch(e){}
try{
  if(window.BroadcastChannel){
    var bcTheme = new BroadcastChannel('omar_theme_v28');
    bcTheme.onmessage = function(ev){ if(ev.data && ev.data.t) window.applyTheme(ev.data.t); };
    var bcFont = new BroadcastChannel('omar_font_v28');
    bcFont.onmessage = function(ev){ if(ev.data && ev.data.size) window.applyFontSize(ev.data.size); };
  }
}catch(e){}


// --- POLLING for SPCK WebView where storage event may not fire ---
(function(){
  var lastFont = localStorage.getItem('omar_font_size') || '13';
  var lastTheme = localStorage.getItem('omar_theme') || 'forest';
  setInterval(function(){
    try{
      var curFont = localStorage.getItem('omar_font_size');
      if(curFont && curFont !== lastFont){
        lastFont = curFont;
        if(window.applyFontSize) window.applyFontSize(curFont);
      }
      var curTheme = localStorage.getItem('omar_theme');
      if(curTheme && curTheme !== lastTheme){
        lastTheme = curTheme;
        if(window.applyTheme) window.applyTheme(curTheme);
      }
    }catch(e){}
  }, 300);
})();
